# --- Force static runtime library to match LLVM/Clang ---
# LLVM/Clang libraries are built with static runtime (/MT), so we must match
# This prevents LNK2038 errors about RuntimeLibrary mismatch
if(MSVC)
    # Remove /MD and /MDd flags and replace with /MT and /MTd
    foreach(flag_var CMAKE_CXX_FLAGS CMAKE_CXX_FLAGS_RELEASE CMAKE_CXX_FLAGS_DEBUG 
                     CMAKE_CXX_FLAGS_RELWITHDEBINFO CMAKE_CXX_FLAGS_MINSIZEREL)
        string(REPLACE "/MD" "/MT" ${flag_var} "${${flag_var}}")
        string(REPLACE "/MDd" "/MTd" ${flag_var} "${${flag_var}}")
    endforeach()
    # Set default runtime library (for CMake 3.15+)
    if(POLICY CMP0091)
        cmake_policy(SET CMP0091 NEW)
        set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
endif()

if(DIA_SDK_FOUND)
    # Add to linker search paths before finding LLVM

    # Also set as environment variable for linker

    # LLVM hardcodes: C:\Program Files (x86)\Microsoft Visual Studio\2019\Professional\DIA SDK

    get_filename_component(DIA_SDK_ROOT "${DIA_SDK_ROOT}" PATH)
    
    set(EXPECTED_DIA_ROOT "C:/Program Files (x86)/Microsoft Visual Studio/2019/Professional/DIA SDK")
    set(EXPECTED_DIA_LIB "${EXPECTED_DIA_ROOT}/lib/amd64/diaguids.lib")
    
    # If DIA SDK is found but not at the expected Professional location, create a junction
    if(NOT EXISTS "${EXPECTED_DIA_LIB}" AND EXISTS "${DIA_SDK_LIB_DIR}/diaguids.lib" AND WIN32)
        message(STATUS "DIA SDK found at: ${DIA_SDK_ROOT}")
        message(STATUS "LLVM expects it at: ${EXPECTED_DIA_ROOT}")
        message(STATUS "Attempting to create directory junction...")
        
        # Get parent directory of expected location
        get_filename_component(EXPECTED_PARENT "${EXPECTED_DIA_ROOT}" PATH)
        
        # Convert paths to native Windows format (backslashes) for Windows commands
        file(TO_NATIVE_PATH "${EXPECTED_PARENT}" EXPECTED_PARENT_NATIVE)
        file(TO_NATIVE_PATH "${EXPECTED_DIA_ROOT}" EXPECTED_DIA_ROOT_NATIVE)
        file(TO_NATIVE_PATH "${DIA_SDK_ROOT}" DIA_SDK_ROOT_NATIVE)
        
        # Try to create parent directory structure if it doesn't exist (non-fatal)
        if(NOT EXISTS "${EXPECTED_PARENT}")
            message(STATUS "Attempting to create parent directory: ${EXPECTED_PARENT}")
            execute_process(
                COMMAND powershell -Command "New-Item -ItemType Directory -Path '${EXPECTED_PARENT_NATIVE}' -Force"
                RESULT_VARIABLE DIR_CREATE_RESULT
                OUTPUT_QUIET
                ERROR_QUIET
            )
            if(NOT DIR_CREATE_RESULT EQUAL 0)
                message(WARNING "Could not create parent directory (requires Administrator privileges)")
                message(WARNING "CMake will continue, but you'll need to create the junction manually")
            endif()
        endif()
        
        # Try to create junction (mklink can create parent directories if needed)
        # First check if junction already exists
        if(NOT EXISTS "${EXPECTED_DIA_ROOT}")
            message(STATUS "Attempting to create junction: ${EXPECTED_DIA_ROOT} -> ${DIA_SDK_ROOT}")
            
            # Try PowerShell first (may work without admin, and can create parent dirs)
            execute_process(
                COMMAND powershell -Command "$parent = Split-Path '${EXPECTED_DIA_ROOT_NATIVE}'; if (-not (Test-Path $parent)) { New-Item -ItemType Directory -Path $parent -Force | Out-Null }; New-Item -ItemType Junction -Path '${EXPECTED_DIA_ROOT_NATIVE}' -Target '${DIA_SDK_ROOT_NATIVE}' -Force"
                RESULT_VARIABLE JUNCTION_RESULT
                OUTPUT_QUIET
                ERROR_QUIET
            )
            
            if(JUNCTION_RESULT EQUAL 0 AND EXISTS "${EXPECTED_DIA_ROOT}")
                message(STATUS "Successfully created junction!")
            else()
                # If PowerShell fails, try cmd mklink (requires admin, but can create parent dirs)
                message(STATUS "PowerShell junction creation failed, trying cmd mklink...")
                # Create parent directory first using cmd
                execute_process(
                    COMMAND cmd /c "if not exist \"${EXPECTED_PARENT_NATIVE}\" mkdir \"${EXPECTED_PARENT_NATIVE}\""
                    RESULT_VARIABLE MKDIR_RESULT
                    OUTPUT_QUIET
                    ERROR_QUIET
                )
                # Now try mklink
                execute_process(
                    COMMAND cmd /c "mklink /J \"${EXPECTED_DIA_ROOT_NATIVE}\" \"${DIA_SDK_ROOT_NATIVE}\""
                    RESULT_VARIABLE MKLINK_RESULT
                    OUTPUT_QUIET
                    ERROR_QUIET
                )
                
                if(MKLINK_RESULT EQUAL 0 AND EXISTS "${EXPECTED_DIA_ROOT}")
                    message(STATUS "Successfully created junction using mklink!")
                else()
                    message(WARNING "Failed to create junction automatically (requires Administrator privileges).")
                    message(WARNING "CMake configuration will continue, but linking may fail.")
                    message(WARNING "")
                    message(WARNING "To fix this, run as Administrator:")
                    message(WARNING "  mklink /J \"${EXPECTED_DIA_ROOT_NATIVE}\" \"${DIA_SDK_ROOT_NATIVE}\"")
                    message(WARNING "")
                    message(WARNING "Or copy the folder instead:")
                    message(WARNING "  xcopy /E /I \"${DIA_SDK_ROOT_NATIVE}\" \"${EXPECTED_DIA_ROOT_NATIVE}\"")
                endif()
            endif()
        else()
            message(STATUS "Junction or directory already exists at expected location")
        endif()
    endif()
else()
